## 선택한 문장의 상위 설명을 질문에 포함합니다

Compound의 `src/modules/news/ask.ts`는 선택한 문장과 그 상위 문장을 노트 순서대로 모읍니다. 이미 선택한 문장은 중복을 줄이되, 하위 문장의 의미를 설명하는 상위 관계를 유지합니다.

```typescript
export function contextLines(lines: Line[], picked: PickedLine[]): ContextLine[] {
  const pos = new Map(lines.map((l, i) => [l.id, i]));
  const up = (i: number) => {
    const out: number[] = [];
    for (let j = i - 1, d = lines[i].depth; j >= 0 && d > 1; j--) {
      if (lines[j].depth < d) {
        out.push(j);
        d = lines[j].depth;
      }
    }
    return out;
  };
  const whole = new Set(picked.filter((p) => p.whole).map((p) => p.id));
  const keep = new Set<number>();
  for (const p of picked) {
    const i = pos.get(p.id);
    if (i === undefined) continue;
    if (!p.whole) keep.add(i);
    for (const j of up(i)) if (!whole.has(lines[j].id)) keep.add(j);
  }
  for (const i of [...keep]) for (const j of up(i)) keep.add(j);
  const idx = [...keep].sort((a, b) => a - b);
  const top = Math.min(...idx.map((i) => lines[i].depth));
  return idx.map((i) => ({ level: lines[i].depth - top, text: plain(lines[i].text).replace(/\s+/g, ' ') }));
}
```

구성한 문맥은 노트의 날짜·항목·선택한 부분·사용자의 질문과 함께 전달합니다.

```typescript
export function buildPrompt(t: AskTarget, question: string) {

}
```

```typescript
- ${mode === 'instant' ? 'After each of my turns, give one short correction if I need it, then continue.' : 'Do not correct me during the conversation. Only if you cannot understand me, say what you think I mean and ask if that is right.'}
- If I say "hint", give me a sentence starter. If I say "hint" again, give a full sample reply. "slower" and "repeat" always work.
- When I say or type "report", stop the role-play and write the report below as text in one message. Use only what I actually said and do not invent sentences. Do not score pronunciation, because you only see a transcript.

Report format (keep the markers exactly, one item per line, columns separated by |):
###REPORT v1
[MISSIONS]
- 1 | done or not | short quote of what I said
[CORRECTIONS]
- what I said | better version | reason in Korean, one line | grammar, vocabulary, word order or naturalness
[BETTER]
- correct but unnatural sentence I said | more natural version
[SAVE]
- expression | Korean meaning | example sentence
[FOCUS]
one thing to focus on next time, in Korean
###END

For [CORRECTIONS] give 3 to 5 items, the ones that hurt understanding most first, then the most frequent. For [SAVE] give 5 useful expressions.`;
}
```

```python
def _reuse_source_versions(c, rid, batch, fid, rows):
    """Validate reuse against the current original rows before inserting any new round."""
    reused = [p for p in rows if p.get("auditor") == "reuse-source" or "reuse_verdict_id" in p]
    if not reused:
        return
    originals = [r for r in _vrows(c, rid, batch) if r["fact_id"] == fid]
    allowed = {r["reuse_verdict_id"]: r for r in _reusable_source_rows(c, rid, fid, batch, originals)}
    for p in reused:
        ident = p.get("reuse_verdict_id")
        expected = allowed.get(ident) if type(ident) is int else None
        if expected is None or p.get("use") or p.get("_use_ref"):
            raise Reject("재사용 원본 판정이 현재 사실·run·묶음·값 판과 맞지 않는다 — 새 요청을 받는다")
        # The worker may combine rows and add retirement metadata, but cannot rewrite the evidence.
        if any(p.get(k) != value for k, value in expected.items()) or p["_fr"] != expected["fact_rev"]:
            raise Reject("재사용 응답의 사실·원문·판정·본문 판이 원본과 다르다 — 원본 행 그대로 제출한다")
        svid = expected["source_version_id"]
        if svid is not None:
            source = c.execute("SELECT s.url FROM ev_source_version v JOIN ev_source s ON s.id=v.source_id WHERE v.id=?",
                               (svid,)).fetchone()
            url = expected["url"] or c.execute("SELECT url FROM ev_fact WHERE id=?", (fid,)).fetchone()[0]
            if not source or _norm_url(source[0]) != _norm_url(url or ""):
                raise Reject("재사용 본문 판이 원본 출처와 맞지 않는다 — 원문을 다시 확인한다")
        # None is also preserved: an old judgment without an archived body cannot acquire one by reuse.
        p["_reuse_source_version"] = svid
```
